# Minimum version of CMake required
cmake_minimum_required(VERSION 3.13)

# Project name and language (C++)
project(MyDigiviceSim VERSION 1.0 LANGUAGES CXX)

# Set the C++ standard we want to use (17 is good)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# --- Find Required Libraries ---
# Find the SDL2 library that we installed via MSYS2
# This command tells CMake to look for SDL2 and stop if it's not found.
find_package(SDL2 REQUIRED)
# find_package(SDL2_image REQUIRED) # We'll uncomment this later if needed

# --- Subdirectories ---
# Tell CMake to look inside the 'src' folder for more build instructions
add_subdirectory(src)
